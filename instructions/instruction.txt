
Add Useful Indexes for Query Performance
NL queries will heavily filter by date, customer, and product:

-- customer
CREATE INDEX idx_customer_code        ON customer(code);
CREATE INDEX idx_customer_name        ON customer(customer_name);

-- orders
CREATE INDEX idx_orders_customer_id   ON orders(customer_id);
CREATE INDEX idx_orders_delivery_date ON orders(delivery_date);
CREATE INDEX idx_orders_order_status  ON orders(order_status);
CREATE INDEX idx_orders_order_no      ON orders(order_no);

-- product
CREATE INDEX idx_product_code         ON product(product_code);
CREATE INDEX idx_product_name         ON product(product_name);
CREATE INDEX idx_product_status       ON product(status);

-- inventory
CREATE INDEX idx_inventory_product    ON inventory(product_code);
CREATE INDEX idx_inventory_lot        ON inventory(lot_number);



Create a Denormalized Summary View
This is a big one. A single flat view joining all 5 tables makes NL-to-SQL dramatically simpler
and more accurate — the AI generates simpler SQL against the view instead of complex multi-table JOINs:

CREATE OR REPLACE VIEW v_sales_summary AS
SELECT
    -- Sales
    s.id                AS sales_id,
    s.order_date,
    s.delivery_date,
    s.line_no,
    s.quantity,
    s.unit_price,
    s.quantity * s.unit_price   AS line_total,
    s.sales_status,
    s.brand,
    s.origin,
    s.reference_no,

    -- Customer
    c.id                AS customer_id,
    c.code              AS customer_code,
    c.customer_name,
    c.short_name,
    c.activation_status AS customer_status,
    c.credit_status,

    -- Orders
    o.id                AS order_id,
    o.order_no,
    o.order_status,
    o.delivery_method,
    o.grand_total,
    o.discount,
    o.delivery_fees,
    o.total_quantity    AS order_total_quantity,
    o.po_number,

    -- Product
    p.id                AS product_id,
    p.product_name,
    p.product_code,
    p.product_number,
    p.status            AS product_status,
    p.new_arrival,

    -- Inventory
    i.id                AS inventory_id,
    i.on_hand,
    i.available_quantity,
    i.expected_quantity,
    i.base_cost,
    i.sell_for,
    i.lot_number,
    i.receive_date

FROM sales s
LEFT JOIN customer c  ON s.customer_code  = c.code
LEFT JOIN orders  o   ON s.reference_no   = o.order_no
LEFT JOIN product p   ON s.product_code   = p.product_code
LEFT JOIN inventory i ON s.product_code   = i.product_code;


Create a Metadata Table
Store human-readable descriptions of every table and column:
CREATE TABLE schema_metadata (
    id               SERIAL PRIMARY KEY,
    table_name       VARCHAR(100) NOT NULL,
    column_name      VARCHAR(100),           -- NULL means it's a table-level description
    data_type        VARCHAR(50),
    description      TEXT NOT NULL,
    sample_values    TEXT,                   -- comma-separated examples
    is_join_key      BOOLEAN DEFAULT FALSE,
    join_references  VARCHAR(200)            -- e.g. 'orders.order_no'
);

Write a Business Glossary
Define terms your users will say in plain English that map to SQL concepts:

CREATE TABLE business_glossary (
    term         VARCHAR(100) PRIMARY KEY,
    sql_meaning  TEXT NOT NULL
);

INSERT INTO business_glossary VALUES
('revenue',         'SUM(sales.quantity * sales.unit_price)'),
('total sales',     'SUM(sales.quantity * sales.unit_price)'),
('active customers','customers WHERE activation_status = ''ACTIVE'''),
('pending orders',  'orders WHERE order_status = ''PENDING'''),
('void orders',     'orders WHERE order_status = ''VOID'''),
('stock level',     'inventory.available_quantity'),
('low stock',       'inventory WHERE available_quantity < 50'),
('new arrivals',    'product WHERE new_arrival = TRUE'),
('this month',      'WHERE order_date >= date_trunc(''month'', CURRENT_DATE)'),
('last month',      'WHERE order_date BETWEEN date_trunc(''month'', CURRENT_DATE - INTERVAL ''1 month'') AND date_trunc(''month'', CURRENT_DATE) - INTERVAL ''1 day'''),
('this year',       'WHERE order_date >= date_trunc(''year'', CURRENT_DATE)'),
('top customers',   'GROUP BY customer_name ORDER BY SUM(quantity * unit_price) DESC'),
('best sellers',    'GROUP BY product_name ORDER BY SUM(quantity) DESC');
